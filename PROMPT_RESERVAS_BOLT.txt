/*
# Módulo de Reservas de canchas

1. Nuevas tablas
- `court_reservations`: reservas futuras (cancha 0-3, fecha, hora, titular lote/nombre, observación,
  estado reservada|abierta|cerrada|cancelada, session_id del turno que se abrió desde la reserva).
- `court_sessions`: turnos de cancha (número de turno correlativo, cancha, reserva de origen, estado
  abierta|cerrada|cancelada, apertura/cierre, jugadores en JSON con lote/nombre/condición, titular,
  total, medio de pago, venta generada, usuarios que abrieron/cerraron).
- `court_session_items`: consumos del turno (producto, nombre, precio, cantidad).

2. Cambios en tablas existentes
- `sales.court_session_id` (uuid, opcional): indica de qué turno de cancha se originó la venta.
  Índice único: un turno solo puede generar UNA venta.

3. Lógica en el servidor
- `open_court_session(...)`: abre un turno de forma atómica (no permite dos turnos abiertos en la misma cancha)
  y marca la reserva de origen como abierta.
- `close_court_session_free(id, user)`: cierra un turno sin consumos (total $0).
- Trigger en `sales`: al registrar una venta con court_session_id, cierra el turno en la misma operación;
  si el turno ya no está abierto la venta se rechaza (evita doble cobro / doble cierre).

4. Seguridad
- RLS en las 3 tablas; la app no usa login de Supabase, por eso las políticas son para anon y authenticated.
- Los turnos no se pueden modificar directamente: solo por las funciones/trigger anteriores.
- Los consumos solo se pueden agregar/quitar mientras el turno esté abierto.
- Reservas: lectura, alta y modificación (cancelar = cambiar estado). Sin borrado.
*/

CREATE OR REPLACE FUNCTION public.open_court_session(
  p_court smallint, p_reservation_id uuid, p_players jsonb,
  p_titular_name text, p_titular_lot text, p_titular_socio_id uuid, p_user text
) RETURNS uuid
LANGUAGE plpgsql SECURITY DEFINER SET search_path = ''
AS $$
DECLARE
  v_id uuid;
BEGIN
  IF btrim(coalesce(p_titular_name,'')) = '' THEN
    RAISE EXCEPTION 'titular_required';
  END IF;
  IF jsonb_typeof(p_players) <> 'array' OR jsonb_array_length(p_players) NOT BETWEEN 1 AND 4 THEN
    RAISE EXCEPTION 'invalid_players';
  END IF;
  IF EXISTS (SELECT 1 FROM public.court_sessions WHERE court = p_court AND status = 'abierta') THEN
    RAISE EXCEPTION 'court_already_open';
  END IF;

  IF p_reservation_id IS NOT NULL THEN
    PERFORM 1 FROM public.court_reservations WHERE id = p_reservation_id AND status = 'reservada' FOR UPDATE;
    IF NOT FOUND THEN
      RAISE EXCEPTION 'reservation_not_available';
    END IF;
  END IF;

  INSERT INTO public.court_sessions(court, reservation_id, players, titular_name, titular_lot, titular_socio_id, opened_by)
  VALUES (p_court, p_reservation_id, p_players, btrim(p_titular_name), btrim(coalesce(p_titular_lot,'')), p_titular_socio_id, coalesce(p_user,''))
  RETURNING id INTO v_id;

  IF p_reservation_id IS NOT NULL THEN
    UPDATE public.court_reservations
    SET status = 'abierta', session_id = v_id, court = p_court, updated_at = now()
    WHERE id = p_reservation_id;
  END IF;

  RETURN v_id;
EXCEPTION WHEN unique_violation THEN
  RAISE EXCEPTION 'court_already_open';
END;
$$;

CREATE OR REPLACE FUNCTION public.close_court_session_free(p_session_id uuid, p_user text)
RETURNS void
LANGUAGE plpgsql SECURITY DEFINER SET search_path = ''
AS $$
DECLARE
  v_res uuid;
BEGIN
  IF EXISTS (SELECT 1 FROM public.court_session_items WHERE session_id = p_session_id) THEN
    RAISE EXCEPTION 'session_has_items';
  END IF;

  UPDATE public.court_sessions
  SET status = 'cerrada', closed_at = now(), total = 0, closed_by = coalesce(p_user,'')
  WHERE id = p_session_id AND status = 'abierta'
  RETURNING reservation_id INTO v_res;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'session_not_open';
  END IF;

  IF v_res IS NOT NULL THEN
    UPDATE public.court_reservations SET status = 'cerrada', updated_at = now() WHERE id = v_res;
  END IF;
END;
$$;

CREATE OR REPLACE FUNCTION public.close_court_session_on_sale()
RETURNS trigger
LANGUAGE plpgsql SECURITY DEFINER SET search_path = ''
AS $$
DECLARE
  v_res uuid;
BEGIN
  IF NEW.court_session_id IS NULL THEN
    RETURN NEW;
  END IF;

  UPDATE public.court_sessions
  SET status = 'cerrada', closed_at = now(), total = NEW.total,
      payment_method = NEW.payment_method, sale_id = NEW.id, closed_by = coalesce(NEW.user_name,'')
  WHERE id = NEW.court_session_id AND status = 'abierta'
  RETURNING reservation_id INTO v_res;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'court_session_not_open';
  END IF;

  IF v_res IS NOT NULL THEN
    UPDATE public.court_reservations SET status = 'cerrada', updated_at = now() WHERE id = v_res;
  END IF;

  RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS trg_close_court_session_on_sale ON sales;
CREATE TRIGGER trg_close_court_session_on_sale
  BEFORE INSERT ON sales FOR EACH ROW EXECUTE FUNCTION public.close_court_session_on_sale();

REVOKE ALL ON FUNCTION public.open_court_session(smallint, uuid, jsonb, text, text, uuid, text) FROM PUBLIC;

######## MIGRACION 3/5 - nombre: court_guest_charge ########
/*
# Cargo automático de INVITADOS en turnos de cancha

1. Cambios en tablas
- `court_session_items`:
  - `kind` (text, 'producto' | 'invitado', default 'producto'): tipo de ítem del turno.
  - `product_id` pasa a ser opcional (el ítem INVITADO no es un producto de stock).
  - Regla: los ítems 'producto' siempre tienen product_id.
  - Índice único: un solo ítem INVITADO por turno.
- `court_sessions`:
  - `guest_charge_manual` (boolean, default false): se vuelve true cuando el operador modifica o elimina
    el cargo INVITADO a mano. Desde ese momento el sistema ya no lo recalcula.

2. Lógica en el servidor
- `sync_court_guest_charge(session)` (interna): si el cargo sigue en modo automático, deja el ítem INVITADO
  con cantidad = jugadores invitados (precio predeterminado $12.000, o el precio actual si ya existe);
  si no hay invitados, quita el ítem.
- `open_court_session(...)`: igual que antes, y además genera el cargo INVITADO al abrir.
- `update_court_session_players(...)`: permite corregir jugadores de un turno abierto y recalcula el cargo
  si sigue automático.
- `set_court_guest_charge(session, cantidad, precio)`: edición manual (cantidad 0 = eliminar). Marca el
  cargo como manual.
- `close_court_session_free`: ahora permite cerrar sin cobro cuando el total es $0 (aunque haya ítems en $0).

3. Seguridad
- Los ítems INVITADO solo se crean/modifican/eliminan mediante las funciones anteriores:
  las políticas de la app para agregar/modificar/quitar consumos quedan limitadas a ítems 'producto'.
- La función interna de sincronización no puede llamarse desde la app.
*/

DO $$
BEGIN

    WHERE table_schema='public' AND table_name='court_session_items' AND column_name='kind') THEN
    ALTER TABLE court_session_items ADD COLUMN kind text NOT NULL DEFAULT 'producto'
      CHECK (kind IN ('producto','invitado'));
  END IF;
  IF NOT EXISTS (SELECT 1 FROM information_schema.columns
    WHERE table_schema='public' AND table_name='court_sessions' AND column_name='guest_charge_manual') THEN
    ALTER TABLE court_sessions ADD COLUMN guest_charge_manual boolean NOT NULL DEFAULT false;

    ALTER TABLE court_session_items ALTER COLUMN product_id DROP NOT NULL;
    ALTER TABLE court_session_items ADD CONSTRAINT court_session_items_product_required

END $$;

CREATE OR REPLACE FUNCTION public.sync_court_guest_charge(p_session_id uuid)
RETURNS void
LANGUAGE plpgsql SECURITY DEFINER SET search_path = ''
AS $$
DECLARE
  v_manual boolean;
  v_players jsonb;
  v_guests integer;
BEGIN
  SELECT guest_charge_manual, players INTO v_manual, v_players
  FROM public.court_sessions WHERE id = p_session_id AND status = 'abierta';
  IF NOT FOUND OR v_manual THEN
    RETURN;
  END IF;

  SELECT count(*) INTO v_guests
  FROM jsonb_array_elements(v_players) p
  WHERE coalesce((p->>'is_socio')::boolean, false) = false;

  IF v_guests = 0 THEN
    DELETE FROM public.court_session_items WHERE session_id = p_session_id AND kind = 'invitado';
  ELSE
    INSERT INTO public.court_session_items(session_id, product_id, product_name, price, quantity, kind)
    VALUES (p_session_id, NULL, 'INVITADO', 12000, v_guests, 'invitado')
    ON CONFLICT (session_id) WHERE kind = 'invitado'
    DO UPDATE SET quantity = EXCLUDED.quantity;
  END IF;
END;
$$;

REVOKE ALL ON FUNCTION public.sync_court_guest_charge(uuid) FROM PUBLIC;
REVOKE EXECUTE ON FUNCTION public.sync_court_guest_charge(uuid) FROM anon, authenticated;

CREATE OR REPLACE FUNCTION public.find_court_conflict(
  p_court smallint, p_start timestamp, p_end timestamp, p_exclude_id uuid
) RETURNS text
LANGUAGE sql STABLE SECURITY DEFINER SET search_path = ''
AS $$
  SELECT to_char(x.s, 'HH24:MI') || '-' || to_char(x.e, 'HH24:MI')
  FROM (
    SELECT (r.reservation_date + r.start_time) AS s, (r.reservation_date + r.end_time) AS e
    FROM public.court_reservations r
    WHERE r.court = p_court AND r.status IN ('reservada','abierta')
      AND r.id IS DISTINCT FROM p_exclude_id
    UNION ALL
    SELECT (cs.opened_at AT TIME ZONE 'America/Argentina/Buenos_Aires'),
           (coalesce(cs.closed_at, greatest(now(), cs.opened_at) + interval '1 minute') AT TIME ZONE 'America/Argentina/Buenos_Aires')
    FROM public.court_sessions cs
    WHERE cs.court = p_court AND cs.status IN ('abierta','cerrada')
      AND cs.reservation_id IS DISTINCT FROM p_exclude_id
  ) x
  WHERE x.s < p_end AND x.e > p_start
  ORDER BY x.s
  LIMIT 1;
$$;

CREATE OR REPLACE FUNCTION public.check_court_event_overlap()
RETURNS trigger
LANGUAGE plpgsql SECURITY DEFINER SET search_path = ''
AS $$
DECLARE
  v_conflict text;
BEGIN
  IF NEW.status <> 'reservada' THEN
    RETURN NEW;
  END IF;
  PERFORM pg_advisory_xact_lock(hashtext('court_agenda'), NEW.court);
  v_conflict := public.find_court_conflict(
    NEW.court, NEW.reservation_date + NEW.start_time, NEW.reservation_date + NEW.end_time, NEW.id);
  IF v_conflict IS NOT NULL THEN
    RAISE EXCEPTION 'court_busy:%', v_conflict;
  END IF;
  RETURN NEW;
END;
$$;

CREATE OR REPLACE FUNCTION public.save_court_event(
  p_group_id uuid, p_event_type text, p_courts smallint[], p_date date, p_start time, p_end time,
  p_titular_name text, p_titular_lot text, p_title text, p_professor text, p_student text,
  p_notes text, p_user text
) RETURNS uuid
LANGUAGE plpgsql SECURITY DEFINER SET search_path = ''
AS $$
DECLARE
  v_group uuid := coalesce(p_group_id, gen_random_uuid());
  v_courts smallint[];
  v_free_rows uuid[];
  v_new_courts smallint[];
  v_row record;
  v_court smallint;
  i integer;
BEGIN
  IF p_event_type NOT IN ('reserva','clase','torneo','mantenimiento') THEN
    RAISE EXCEPTION 'invalid_event';
  END IF;
  SELECT array_agg(DISTINCT c ORDER BY c) INTO v_courts FROM unnest(p_courts) c WHERE c BETWEEN 0 AND 3;
  IF v_courts IS NULL OR array_length(v_courts, 1) <> coalesce(array_length(p_courts, 1), 0) THEN
    RAISE EXCEPTION 'invalid_courts';
  END IF;
  IF p_event_type IN ('reserva','clase') AND array_length(v_courts, 1) <> 1 THEN
    RAISE EXCEPTION 'invalid_courts';
  END IF;
  IF p_date IS NULL OR p_start IS NULL OR p_end IS NULL OR p_end <= p_start THEN
    RAISE EXCEPTION 'invalid_time';
  END IF;
  IF p_event_type = 'reserva' AND btrim(coalesce(p_titular_name,'')) = '' THEN
    RAISE EXCEPTION 'titular_required';
  END IF;
  IF p_event_type = 'torneo' AND btrim(coalesce(p_title,'')) = '' THEN
    RAISE EXCEPTION 'title_required';
  END IF;

  IF p_group_id IS NOT NULL THEN
    IF NOT EXISTS (SELECT 1 FROM public.court_reservations WHERE event_group = p_group_id AND status = 'reservada') THEN
      RAISE EXCEPTION 'event_not_editable';
    END IF;
    IF EXISTS (SELECT 1 FROM public.court_reservations WHERE event_group = p_group_id AND status = 'abierta') THEN
      RAISE EXCEPTION 'event_not_editable';
    END IF;
  END IF;

  -- Rows whose court is no longer selected are reused for newly selected courts; leftovers are cancelled.
  SELECT coalesce(array_agg(id), '{}') INTO v_free_rows FROM public.court_reservations
  WHERE event_group = v_group AND status = 'reservada' AND NOT (court = ANY (v_courts));
  SELECT coalesce(array_agg(c), '{}') INTO v_new_courts FROM unnest(v_courts) c
  WHERE NOT EXISTS (SELECT 1 FROM public.court_reservations r WHERE r.event_group = v_group AND r.status = 'reservada' AND r.court = c);

  FOR i IN 1 .. coalesce(array_length(v_free_rows, 1), 0) LOOP
    IF i <= coalesce(array_length(v_new_courts, 1), 0) THEN
      UPDATE public.court_reservations SET court = v_new_courts[i] WHERE id = v_free_rows[i];
    ELSE
      UPDATE public.court_reservations SET status = 'cancelada', updated_at = now() WHERE id = v_free_rows[i];
    END IF;
  END LOOP;

  FOR v_row IN SELECT id FROM public.court_reservations WHERE event_group = v_group AND status = 'reservada' LOOP
    UPDATE public.court_reservations SET
      event_type = p_event_type, reservation_date = p_date, start_time = p_start, end_time = p_end,
      titular_name = btrim(coalesce(p_titular_name,'')), titular_lot = btrim(coalesce(p_titular_lot,'')),
      title = btrim(coalesce(p_title,'')), professor = btrim(coalesce(p_professor,'')),
      student = btrim(coalesce(p_student,'')), notes = btrim(coalesce(p_notes,'')), updated_at = now()
    WHERE id = v_row.id;
  END LOOP;

  FOR i IN coalesce(array_length(v_free_rows, 1), 0) + 1 .. coalesce(array_length(v_new_courts, 1), 0) LOOP
    v_court := v_new_courts[i];
    INSERT INTO public.court_reservations(
      court, reservation_date, start_time, end_time, titular_lot, titular_name, notes, status,
      created_by, event_type, event_group, title, professor, student)
    VALUES (
      v_court, p_date, p_start, p_end, btrim(coalesce(p_titular_lot,'')), btrim(coalesce(p_titular_name,'')),
      btrim(coalesce(p_notes,'')), 'reservada', coalesce(p_user,''), p_event_type, v_group,
      btrim(coalesce(p_title,'')), btrim(coalesce(p_professor,'')), btrim(coalesce(p_student,'')));
  END LOOP;

  RETURN v_group;
END;
$$;

CREATE OR REPLACE FUNCTION public.open_court_session(
  p_court smallint, p_reservation_id uuid, p_players jsonb,
  p_titular_name text, p_titular_lot text, p_titular_socio_id uuid, p_user text

BEGIN
  IF btrim(coalesce(p_titular_name,'')) = '' THEN

  INTO v_busy
  FROM public.court_reservations r
  WHERE r.court = p_court AND r.status = 'reservada' AND r.id IS DISTINCT FROM p_reservation_id
    AND (r.reservation_date + r.start_time) <= v_now AND (r.reservation_date + r.end_time) > v_now
  LIMIT 1;
  IF v_busy IS NOT NULL THEN
    RAISE EXCEPTION 'court_busy:%', v_busy;
  END IF;

  INSERT INTO public.court_sessions(court, reservation_id, players, titular_name, titular_lot, titular_socio_id, opened_by)

  IF p_reservation_id IS NOT NULL THEN
    UPDATE public.court_reservations
    SET status = 'abierta', session_id = v_id, updated_at = now()
    WHERE id = p_reservation_id;
  END IF;

  PERFORM public.sync_court_guest_charge(v_id);

  RETURN v_id;
EXCEPTION WHEN unique_violation THEN
  RAISE EXCEPTION 'court_already_open';
END;
$$;

export function preparePayments(
  amounts: PaymentAmounts,
  total: number,
  customerName: string,
  customerLot: string
): { payments: Payment[] } | { error: string } {
  let payments = (Object.keys(EMPTY_PAYMENTS) as PaymentMethod[])
    .map((method) => ({ method, amount: amounts[method] }))
    .filter((p) => p.amount > 0.009);

  if (payments.length === 0) {
    payments = [{ method: 'efectivo', amount: total }];
  }

  const paymentsTotal = payments.reduce((sum, p) => sum + p.amount, 0);
  if (Math.abs(paymentsTotal - total) > 0.01) {
    return {
      error: `La suma de los montos de pago (${paymentsTotal.toFixed(2)}) no coincide con el total (${total.toFixed(2)}).`,
    };
  }

  const hasCuentaCorriente = payments.some((p) => p.method === 'cuenta_corriente');
  if (hasCuentaCorriente && (!customerName.trim() || !customerLot.trim())) {
    return { error: 'Para ventas en cuenta corriente es obligatorio ingresar el nombre y el número de lote del cliente.' };
  }
  if (hasCuentaCorriente && isInvalidCcLot(customerLot)) {
    return { error: LOTE_INCORRECTO_MSG };
  }
  if (payments.some((p) => p.method !== 'efectivo') && (!customerName.trim() || !customerLot.trim())) {
    return { error: 'Para pagos que no son en efectivo debés completar el nombre y el lote del cliente.' };
  }

  return { payments };
}

interface ProcessSaleInput {
  shift: Shift;
  items: SaleItem[];
  payments: Payment[];
  customerName: string;
  customerLot: string;
  courtSessionId?: string;
}

type ProcessSaleResult =
  | { ok: true; saleId: string; saleNumber: string }
  | { ok: false; error: string };

export async function processSale({
  shift,
  items,
  payments,
  customerName,
  customerLot,
  courtSessionId,
}: ProcessSaleInput): Promise<ProcessSaleResult> {
  const name = customerName.trim();
  const lot = customerLot.trim();
  const total = items.reduce((sum, i) => sum + i.subtotal, 0);
  const hasCuentaCorriente = payments.some((p) => p.method === 'cuenta_corriente');

  let storedPaymentMethod: PaymentMethod = payments[0].method;
  if (payments.some((p) => p.method === 'efectivo')) storedPaymentMethod = 'efectivo';
  if (hasCuentaCorriente && payments.length === 1) storedPaymentMethod = 'cuenta_corriente';

  const saleNumber = `V-${Date.now()}`;
  const { data: saleResult, error: saleError } = await supabase
    .from('sales')
    .insert([{
      sale_number: saleNumber,
      user_id: shift.user_id,
      user_name: shift.user_name,
      shift_id: shift.id,
      items,
      subtotal: total,
      discount: 0,
      total,
      payment_method: storedPaymentMethod,
      customer_name: name || null,
      customer_lot: lot || null,
      payments,
      court_session_id: courtSessionId ?? null,
    }])
    .select()
    .single();

  if (saleError || !saleResult) {
    console.error('Error insertando venta:', saleError);
    const msg = saleError?.message ?? '';
    if (msg.includes('court_session_not_open') || msg.includes('uq_sales_court_session')) {
      return { ok: false, error: 'Este turno ya fue cerrado y cobrado. No se generó una nueva venta.' };
    }
    return { ok: false, error: `Error al registrar la venta: ${msg}` };
  }

  const productIds = Array.from(new Set(items.map((i) => i.product_id).filter(Boolean)));
  const { data: stockRows } = await supabase.from('products').select('id, stock').in('id', productIds);
  for (const item of items) {
    const prod = (stockRows || []).find((p) => p.id === item.product_id);
    if (prod) {
      const { error: stockError } = await supabase
        .from('products')
        .update({ stock: prod.stock - item.quantity })
        .eq('id', item.product_id);
      if (stockError) console.error('Error actualizando stock:', stockError);
      prod.stock -= item.quantity;
    }
  }

  const customerSuffix = name || lot ? ` - ${name} (Lote ${lot || '-'})` : '';
  const cashRows = payments
    .filter((p) => p.method !== 'cuenta_corriente')
    .map((p) => ({
      shift_id: shift.id,
      type: 'income',
      category: 'venta',
      amount: p.amount,
      payment_method: p.method,
      description: `Venta ${saleNumber}${customerSuffix}`,
    }));

  if (hasCuentaCorriente) {
    const ccAmount = payments.find((p) => p.method === 'cuenta_corriente')!.amount;
    cashRows.push({
      shift_id: shift.id,
      type: 'income',
      category: 'venta',
      amount: ccAmount,
      payment_method: 'cuenta_corriente',
      description: `Venta ${saleNumber} - ${name} (Lote ${lot || '-'}) [Cuenta Corriente]`,
    });

    const { data: existingAccount } = await supabase
      .from('customer_accounts')
      .select('*')
      .eq('customer_name', name)
      .eq('customer_lot', lot)
      .maybeSingle();

    let accountId: string;
    let currentDebt = 0;

    if (existingAccount) {
      accountId = existingAccount.id;
      currentDebt = Number(existingAccount.total_debt);
    } else {
      const { data: newAccount, error: accountError } = await supabase
        .from('customer_accounts')
        .insert([{ customer_name: name, customer_lot: lot, total_debt: 0 }])
        .select()
        .single();

      if (accountError || !newAccount) {
        console.error('Error creando cuenta corriente:', accountError);
        return { ok: false, error: 'Error al registrar la cuenta corriente.' };
      }
      accountId = newAccount.id;
    }

    const newBalance = currentDebt + ccAmount;

    await supabase
      .from('customer_accounts')
      .update({ total_debt: newBalance, updated_at: new Date().toISOString() })
      .eq('id', accountId);

    await supabase.from('current_account_transactions').insert([{
      customer_account_id: accountId,
      customer_name: name,
      customer_lot: lot,
      type: 'cargo',
      amount: ccAmount,
      balance_after: newBalance,
      sale_id: saleResult.id,
      sale_number: saleNumber,
      description: `Venta ${saleNumber}`,
      user_name: shift.user_name,
      shift_id: shift.id,
    }]);
  }

  if (cashRows.length > 0) {
    const { error: cashError } = await supabase.from('cash_transactions').insert(cashRows);
    if (cashError) {
      console.error('Error insertando caja:', cashError);
      return { ok: false, error: 'La venta se registró, pero hubo un error al registrar el movimiento en caja.' };
    }
  }

  return { ok: true, saleId: saleResult.id, saleNumber };
}

  const initialStart = event ? formatTime(event.start_time) : defaultTime ?? '18:00';

  const [courts, setCourts] = useState<number[]>([event?.court ?? defaultCourt]);
  const [date, setDate] = useState(event?.reservation_date ?? defaultDate ?? todayISO());
  const [start, setStart] = useState(initialStart);
  const [end, setEnd] = useState(
    event ? formatTime(event.end_time) : hhmmOf(minutesOf(initialStart) + DEFAULT_DURATION[type])
  );
  const [titular, setTitular] = useState<CourtPlayer>({
    lot: event?.titular_lot ?? '',
    name: event?.titular_name ?? '',
    socio_id: null,
    is_socio: false,
  });
  const [title, setTitle] = useState(event?.title ?? '');
  const [professor, setProfessor] = useState(event?.professor ?? '');
  const [student, setStudent] = useState(event?.student ?? '');
  const [notes, setNotes] = useState(event?.notes ?? '');

    if (!event || !multiCourt) return;
    supabase
      .from('court_reservations')
      .select('court')
      .eq('event_group', event.event_group)

    if (value && end && minutesOf(end) > minutesOf(start)) {
      setEnd(hhmmOf(minutesOf(value) + minutesOf(end) - minutesOf(start)));

    setStart(value);
  };

    setError('');
    if (!courts.length) return setError('Seleccione al menos una cancha.');
    if (!date || !start || !end) return setError('Complete fecha, hora desde y hora hasta.');
    if (minutesOf(end) <= minutesOf(start)) return setError('La hora "hasta" debe ser posterior a la hora "desde".');
    if (type === 'reserva' && !titular.name.trim()) return setError('Ingrese el nombre del titular.');
    if (type === 'torneo' && !title.trim()) return setError('Ingrese el nombre del torneo.');
    setSaving(true);
    const { error: rpcError } = await supabase.rpc('save_court_event', {
      p_group_id: event?.event_group ?? null,
      p_event_type: type,
      p_courts: courts,
      p_date: date,
      p_start: start,
      p_end: end,
      p_titular_name: type === 'reserva' ? titular.name : '',
      p_titular_lot: type === 'reserva' ? titular.lot : '',
      p_title: multiCourt ? title : '',
      p_professor: type === 'clase' ? professor : '',
      p_student: type === 'clase' ? student : '',
      p_notes: notes,
      p_user: userName,

import { Loader2, PlayCircle, Save, Users, X } from 'lucide-react';
import { supabase, CourtPlayer, CourtReservation, CourtSession } from '../../lib/supabase';
import LotPersonInput from './LotPersonInput';
import { courtLabel, rpcErrorMessage } from './reservasUtils';

interface OpenCourtModalProps {
  court: number;
  reservation?: CourtReservation;
  session?: CourtSession;
  userName: string;
  onClose: () => void;
  onDone: (sessionId: string) => void;
}

const emptyPlayer = (): CourtPlayer => ({ lot: '', name: '', socio_id: null, is_socio: false });

export default function OpenCourtModal({ court, reservation, session, userName, onClose, onDone }: OpenCourtModalProps) {
  const [players, setPlayers] = useState<CourtPlayer[]>(
    session
      ? [0, 1, 2, 3].map((i) => session.players[i] ?? emptyPlayer())
      : [
          reservation
            ? { lot: reservation.titular_lot, name: reservation.titular_name, socio_id: null, is_socio: false }
            : emptyPlayer(),
          emptyPlayer(),
          emptyPlayer(),
          emptyPlayer(),
        ]
  );
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState('');

  const updatePlayer = (index: number, player: CourtPlayer) =>
    setPlayers((prev) => prev.map((p, i) => (i === index ? player : p)));

  const handleOpen = async () => {
    setError('');
    const titular = players[0];
    if (!titular.name.trim()) return setError('El Jugador 1 (titular) es obligatorio.');
    const incomplete = players.findIndex((p) => p.lot.trim() && !p.name.trim());
    if (incomplete >= 0) return setError(`Complete el nombre del Jugador ${incomplete + 1} o borre su lote.`);

    const filled = players
      .filter((p) => p.name.trim())
      .map((p) => ({ ...p, lot: p.lot.trim(), name: p.name.trim() }));

    setSaving(true);
    const titularFields = {
      p_players: filled,
      p_titular_name: titular.name.trim(),
      p_titular_lot: titular.lot.trim(),
      p_titular_socio_id: titular.is_socio ? titular.socio_id : null,
    };

    if (session) {
      const { error: rpcError } = await supabase.rpc('update_court_session_players', {
        p_session_id: session.id,
        ...titularFields,
      });
      setSaving(false);
      if (rpcError) return setError(rpcErrorMessage(rpcError.message));
      onDone(session.id);
      return;
    }

    const { data, error: rpcError } = await supabase.rpc('open_court_session', {
      p_court: court,
      p_reservation_id: reservation?.id ?? null,
      ...titularFields,
      p_user: userName,
    });
    setSaving(false);
    if (rpcError || !data) return setError(rpcErrorMessage(rpcError?.message));
    onDone(data as string);
  };

  return (
    <div className="fixed inset-0 bg-slate-900/60 backdrop-blur-sm flex items-center justify-center z-50 p-4">
      <div className="bg-white rounded-2xl shadow-2xl w-full max-w-2xl max-h-[92vh] flex flex-col">
        <div className="flex items-center justify-between px-6 py-4 border-b border-slate-100">
          <div className="flex items-center gap-3">
            <div className="p-2 rounded-xl bg-emerald-100 text-emerald-700">
              {session ? <Users size={20} /> : <PlayCircle size={20} />}
            </div>
            <div>
              <h3 className="text-lg font-bold text-slate-800">
                {session ? `Jugadores - ${courtLabel(court)}` : `Abrir ${courtLabel(court)}`}
              </h3>
              <p className="text-xs text-slate-500">Hasta 4 jugadores. El Jugador 1 es el titular.</p>
            </div>
          </div>
          <button onClick={onClose} className="p-2 rounded-lg hover:bg-slate-100 text-slate-500">
            <X size={20} />
          </button>
        </div>

        <div className="p-6 grid grid-cols-1 md:grid-cols-2 gap-3 overflow-y-auto">
          {players.map((p, i) => (
            <LotPersonInput
              key={i}
              label={i === 0 ? 'Jugador 1 - Titular' : `Jugador ${i + 1}`}
              value={p}
              onChange={(player) => updatePlayer(i, player)}
              highlight={i === 0}
            />
          ))}
        </div>

        <div className="px-6 py-4 border-t border-slate-100 space-y-3">
          {error && (
            <div className="bg-red-50 border border-red-200 text-red-700 rounded-xl px-4 py-3 text-sm font-medium">{error}</div>
          )}
          <button
            onClick={handleOpen}
            disabled={saving}
            className="w-full flex items-center justify-center gap-2 py-3.5 rounded-xl bg-emerald-600 hover:bg-emerald-700 disabled:bg-slate-300 text-white font-bold shadow-lg transition-colors"

            {session ? 'GUARDAR JUGADORES' : 'ABRIR CANCHA'}
          </button>
        </div>
      </div>
    </div>
  );

interface CloseTurnoModalProps {

      !window.confirm(`Esta deuda se cargará a:\n${session.titular_name}\nLote ${session.titular_lot}\n¿Confirmar?`)

    setSession(data as CourtSession);

  useEffect(() => {
    loadSession();
    loadItems();
    loadProducts();
    const timer = setInterval(() => setTick((t) => t + 1), 30000);
    return () => clearInterval(timer);
  }, [loadSession, loadItems, loadProducts]);

  const isOpen = session?.status === 'abierta';
  const total = items.reduce((s, i) => s + Number(i.price) * i.quantity, 0);
  const guestItem = items.find((i) => i.kind === 'invitado');
  const productItems = items.filter((i) => i.kind !== 'invitado');

  const qtyInTurno = (productId: string) =>
    items.filter((i) => i.product_id === productId).reduce((s, i) => s + i.quantity, 0);

  const filtered = useMemo(
    () =>
      products.filter(
        (p) => p.stock > 0 && !isGuestProduct(p.name) && p.name.toLowerCase().includes(search.toLowerCase())
      ),
    [products, search]
  );

  const saleItems: SaleItem[] = useMemo(() => {
    const map = new Map<string, SaleItem>();
    for (const i of items) {
      const price = Number(i.price);
      const key = `${i.kind}|${i.product_id}|${price}`;
      const prev = map.get(key);
      map.set(key, {
        product_id: i.product_id ?? '',
        product_name: i.product_name,
        price,
        quantity: (prev?.quantity ?? 0) + i.quantity,
        subtotal: (prev?.subtotal ?? 0) + price * i.quantity,
      });
    }
    return Array.from(map.values());
  }, [items]);

  const withBusy = async (fn: () => Promise<void>) => {
    if (busy) return;
    setBusy(true);
    await fn();
    setBusy(false);
  };

  const addProduct = (product: Product) =>
    withBusy(async () => {
      const fresh = await supabase.from('products').select('stock, price').eq('id', product.id).maybeSingle();
      const stock = fresh.data?.stock ?? product.stock;
      const price = Number(fresh.data?.price ?? product.price);
      if (qtyInTurno(product.id) + 1 > stock) {
        alert('Stock insuficiente');
        return;
      }
      const existing = items.find((i) => i.product_id === product.id && Number(i.price) === price);
      const { error } = existing
        ? await supabase.from('court_session_items').update({ quantity: existing.quantity + 1 }).eq('id', existing.id)
        : await supabase.from('court_session_items').insert([
            { session_id: sessionId, product_id: product.id, product_name: product.name, price, quantity: 1 },
          ]);
      if (error) alert('No se pudo agregar el producto. Verifique que el turno siga abierto.');
      await Promise.all([loadItems(), loadProducts()]);
    });

  const changeQty = (item: CourtSessionItem, qty: number) =>
    withBusy(async () => {
      if (qty > item.quantity) {
        const product = products.find((p) => p.id === item.product_id);
        if (product && qtyInTurno(product.id) + 1 > product.stock) {
          alert('Stock insuficiente');
          return;
        }
      }
      const { error } =
        qty <= 0
          ? await supabase.from('court_session_items').delete().eq('id', item.id)
          : await supabase.from('court_session_items').update({ quantity: qty }).eq('id', item.id);
      if (error) alert('No se pudo modificar el consumo.');
      await loadItems();
    });

  const setGuestCharge = (quantity: number, price: number) =>
    withBusy(async () => {
      const { error } = await supabase.rpc('set_court_guest_charge', {
        p_session_id: sessionId,
        p_quantity: Math.max(0, quantity),
        p_price: price,
      });
      if (error) alert(rpcErrorMessage(error.message));
      await Promise.all([loadItems(), loadSession()]);
    });

  const handleCloseTurno = () =>
    withBusy(async () => {
      if (!session) return;
      await loadItems();
      const { data: latest } = await supabase
        .from('court_session_items')
        .select('price, quantity')
        .eq('session_id', sessionId);
      const latestTotal = (latest || []).reduce((s, i) => s + Number(i.price) * i.quantity, 0);
      if (latestTotal > 0) {
        if (!shift) {
          alert('No hay un turno de caja abierto. Abra la caja para poder cobrar.');
          return;
        }
        setShowCheckout(true);
        return;
      }
      if (!window.confirm('El turno no tiene consumos ($0). ¿Cerrar la cancha?')) return;
      const { error } = await supabase.rpc('close_court_session_free', {
        p_session_id: session.id,
        p_user: shift?.user_name ?? '',
      });
      if (error) {
        alert(rpcErrorMessage(error.message));
        await loadSession();
        return;
      }
      onClosed();
    });

  if (loadError) {
    return (
      <div className="bg-white rounded-2xl p-8 text-center space-y-4">
        <p className="text-red-600 font-semibold">{loadError}</p>
        <button onClick={onBack} className="px-4 py-2 rounded-xl bg-slate-800 text-white">Volver</button>
      </div>
    );
  }

  if (!session) {
    return (
      <div className="flex justify-center py-20">
        <Loader2 className="animate-spin text-slate-400" size={32} />
      </div>
    );
  }

  return (
    <div className="space-y-5">
      <div className="flex flex-wrap items-center gap-3 justify-between">
        <button
          onClick={onBack}
          className="flex items-center gap-2 px-4 py-2.5 rounded-xl bg-white border border-slate-200 hover:bg-slate-50 font-semibold text-slate-700 shadow-sm"
        >
          <ArrowLeft size={18} /> Canchas
        </button>
        <div className="flex items-center gap-3">
          <span className="text-sm text-slate-500">Turno #{session.session_number}</span>
          <span
            className={`px-3 py-1 rounded-full text-xs font-bold ${
              isOpen ? 'bg-amber-100 text-amber-800' : 'bg-slate-200 text-slate-700'
            }`}
          >
            {isOpen ? 'EN JUEGO' : session.status.toUpperCase()}
          </span>
        </div>
      </div>

      <div className="bg-gradient-to-br from-slate-900 to-slate-800 text-white rounded-2xl p-5 shadow-xl grid grid-cols-2 md:grid-cols-4 gap-4">
        <div>
          <p className="text-xs uppercase tracking-wide text-slate-400">Cancha</p>
          <p className="text-2xl font-bold">{courtLabel(session.court)}</p>
        </div>
        <div>
          <p className="text-xs uppercase tracking-wide text-slate-400">Apertura</p>
          <p className="text-2xl font-bold">{formatClock(session.opened_at)}</p>
        </div>
        <div>
          <p className="text-xs uppercase tracking-wide text-slate-400">Tiempo</p>
          <p className="text-2xl font-bold flex items-center gap-2">
            <Clock size={18} className="text-amber-400" />
            {formatDuration(session.opened_at, session.closed_at)}
          </p>
        </div>
        <div>
          <p className="text-xs uppercase tracking-wide text-slate-400">Consumo</p>
          <p className="text-2xl font-bold text-emerald-400">{money(total)}</p>
        </div>
      </div>

      <div className="grid grid-cols-1 lg:grid-cols-3 gap-5">
        <div className="lg:col-span-2 space-y-4">
          <div className="bg-white rounded-2xl border border-slate-200 p-4 shadow-sm">
            <div className="flex items-center justify-between mb-3">
              <h4 className="text-sm font-bold text-slate-700">Jugadores</h4>
              {isOpen && (
                <button
                  onClick={() => setShowPlayers(true)}
                  className="flex items-center gap-1.5 px-3 py-1.5 rounded-lg text-xs font-semibold text-sky-700 hover:bg-sky-50 border border-sky-200"
                >
                  <Users size={14} /> Editar jugadores
                </button>
              )}
            </div>
            <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
              {session.players.map((p, i) => (
                <div
                  key={i}
                  className={`flex items-center justify-between rounded-xl px-3 py-2.5 border-2 ${
                    p.is_socio ? 'border-emerald-300 bg-emerald-50' : 'border-red-200 bg-red-50'
                  }`}
                >
                  <div className="min-w-0">
                    <p className="text-[11px] font-bold uppercase text-slate-500">
                      Jugador {i + 1}
                      {i === 0 && ' · Titular'}
                    </p>
                    <p className="font-semibold text-slate-800 truncate">{p.name}</p>
                    <p className="text-xs text-slate-500">Lote {p.lot || '-'}</p>
                  </div>
                  {p.is_socio ? (
                    <span className="flex items-center gap-1 text-xs font-bold text-emerald-700">
                      <CheckCircle2 size={14} /> SOCIO
                    </span>
                  ) : (
                    <span className="flex items-center gap-1 text-xs font-bold text-red-700">
                      <UserX size={14} /> INVITADO
                    </span>
                  )}
                </div>
              ))}
            </div>
          </div>

          {isOpen && (
            <div className="bg-white rounded-2xl border border-slate-200 p-4 shadow-sm space-y-3">
              <div className="relative">
                <Search className="absolute left-3 top-1/2 -translate-y-1/2 text-slate-400" size={18} />
                <input
                  type="text"
                  placeholder="Buscar producto..."
                  value={search}
                  onChange={(e) => setSearch(e.target.value)}
                  className="w-full pl-10 pr-4 py-2.5 bg-slate-50 border border-slate-200 rounded-xl focus:ring-2 focus:ring-emerald-500"
                />
              </div>
              <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 gap-2 max-h-[360px] overflow-y-auto pr-1">
                {filtered.map((p) => (
                  <button
                    key={p.id}
                    onClick={() => addProduct(p)}
                    disabled={busy}
                    className="text-left p-3 rounded-xl border-2 border-slate-200 hover:border-emerald-500 hover:shadow-lg bg-gradient-to-br from-white to-slate-50 transition-all active:scale-95 disabled:opacity-60"
                  >
                    <p className="font-bold text-sm text-slate-800 leading-tight">{p.name}</p>
                    <p className="text-lg font-bold text-emerald-600 mt-1">{money(p.price)}</p>
                    <p className="text-[10px] text-slate-500">Stock: {p.stock}</p>
                  </button>
                ))}
                {filtered.length === 0 && (
                  <p className="col-span-full text-center text-sm text-slate-400 py-6">No hay productos con stock.</p>
                )}
              </div>
            </div>
          )}
        </div>

        <div className="bg-white rounded-2xl border-2 border-emerald-200 p-5 shadow-xl h-fit space-y-4">
          <h4 className="text-lg font-bold text-slate-800">Consumos</h4>
          <div className="space-y-2 max-h-[340px] overflow-y-auto">
            {items.length === 0 && <p className="text-sm text-slate-400 text-center py-6">Sin consumos</p>}
            {guestItem && (
              <GuestChargeRow
                item={guestItem}
                manual={session.guest_charge_manual}
                editable={isOpen}
                busy={busy}
                onSet={setGuestCharge}
              />
            )}
            {productItems.map((item) => (
              <div key={item.id} className="rounded-xl border border-slate-200 p-3">
                <div className="flex justify-between gap-2 text-sm">
                  <span className="font-semibold text-slate-800">
                    {item.product_name} x{item.quantity}
                  </span>
                  <span className="font-bold text-slate-800">{money(Number(item.price) * item.quantity)}</span>
                </div>
                {isOpen && (
                  <div className="flex items-center justify-between mt-2">
                    <div className="flex items-center gap-2">
                      <button
                        onClick={() => changeQty(item, item.quantity - 1)}
                        disabled={busy}
                        className="w-8 h-8 rounded-lg bg-slate-100 hover:bg-slate-200 flex items-center justify-center"
                      >
                        <Minus size={14} />
                      </button>
                      <span className="w-6 text-center font-bold">{item.quantity}</span>
                      <button
                        onClick={() => changeQty(item, item.quantity + 1)}
                        disabled={busy}
                        className="w-8 h-8 rounded-lg bg-slate-100 hover:bg-slate-200 flex items-center justify-center"
                      >
                        <Plus size={14} />
                      </button>
                    </div>
                    <button
                      onClick={() => changeQty(item, 0)}
                      disabled={busy}
                      className="p-2 rounded-lg text-red-500 hover:bg-red-50"
                    >
                      <Trash2 size={16} />
                    </button>
                  </div>
                )}
              </div>
            ))}
          </div>
          {isOpen && !guestItem && (
            <button
              onClick={() => setGuestCharge(1, Number(products.find((p) => isGuestProduct(p.name))?.price ?? 12000))}
              disabled={busy}
              className="w-full flex items-center justify-center gap-2 py-2 rounded-xl border-2 border-dashed border-red-200 text-red-700 hover:bg-red-50 text-sm font-semibold"
            >
              <UserPlus size={16} /> Agregar cargo INVITADO
            </button>
          )}
          <div className="flex justify-between items-center border-t-2 border-emerald-200 pt-4">
            <span className="text-xl font-bold text-slate-800">TOTAL</span>
            <span className="text-3xl font-bold text-emerald-600">{money(total)}</span>
          </div>
          {isOpen ? (
            <button
              onClick={handleCloseTurno}
              disabled={busy}
              className="w-full flex items-center justify-center gap-2 py-4 rounded-xl bg-slate-900 hover:bg-slate-800 disabled:bg-slate-400 text-white font-bold shadow-lg transition-colors"
            >
              <Lock size={20} /> CERRAR TURNO
            </button>
          ) : (
            <p className="text-center text-sm text-slate-500">Turno cerrado {session.closed_at && `a las ${formatClock(session.closed_at)}`}</p>
          )}
        </div>
      </div>

      {showPlayers && (
        <OpenCourtModal
          court={session.court}
          session={session}
          userName={shift?.user_name ?? ''}
          onClose={() => setShowPlayers(false)}
          onDone={async () => {
            setShowPlayers(false);
            await Promise.all([loadSession(), loadItems()]);
          }}
        />
      )}

      {showCheckout && shift && (
        <CloseTurnoModal
          session={session}
          items={saleItems}
          shift={shift}
          onClose={() => setShowCheckout(false)}
          onPaid={(saleNumber) => {
            setShowCheckout(false);
            alert(`Venta ${saleNumber} completada. Turno cerrado.`);
            onClosed();
          }}
        />
      )}
    </div>
  );
}

import { useEffect, useState } from 'react';
import { CalendarCheck, Clock, Loader2, Receipt, ShoppingBag, Users, X } from 'lucide-react';
import { supabase, CourtReservation, CourtSession, CourtSessionItem } from '../../lib/supabase';
import { PAYMENT_LABELS, Payment } from '../../lib/processSale';
import { EVENT_META, courtLabel, formatClock, formatDateTime, formatDuration, formatTime, money } from './reservasUtils';

interface SaleDetail {
  sale_number: string;
  created_at: string;
  user_name: string;
  total: number;
  payments: Payment[] | null;
  payment_method: Payment['method'];
  customer_name: string | null;
  customer_lot: string | null;
}

interface Detail {
  session: CourtSession;
  items: CourtSessionItem[];
  reservation: CourtReservation | null;
  sale: SaleDetail | null;
}

const sectionTitle = 'flex items-center gap-2 text-xs font-bold uppercase tracking-wide text-slate-500 mb-3';

export default function TurnoDetailModal({ sessionId, onClose }: { sessionId: string; onClose: () => void }) {
  const [detail, setDetail] = useState<Detail | null>(null);
  const [error, setError] = useState('');

  useEffect(() => {
    (async () => {
      const { data: session, error: sErr } = await supabase.from('court_sessions').select('*').eq('id', sessionId).maybeSingle();
      if (sErr || !session) {
        setError('No se pudo cargar el detalle del turno.');
        return;
      }
      const s = session as CourtSession;
      const [itemsRes, resRes, saleRes] = await Promise.all([
        supabase.from('court_session_items').select('*').eq('session_id', s.id).order('created_at'),
        s.reservation_id
          ? supabase.from('court_reservations').select('*').eq('id', s.reservation_id).maybeSingle()
          : Promise.resolve({ data: null, error: null }),
        s.sale_id
          ? supabase
              .from('sales')
              .select('sale_number, created_at, user_name, total, payments, payment_method, customer_name, customer_lot')
              .eq('id', s.sale_id)
              .maybeSingle()
          : Promise.resolve({ data: null, error: null }),
      ]);
      if (itemsRes.error || resRes.error || saleRes.error) {
        setError('No se pudo cargar el detalle del turno.');
        return;
      }
      setDetail({
        session: s,
        items: (itemsRes.data as CourtSessionItem[]) || [],
        reservation: (resRes.data as CourtReservation | null) ?? null,
        sale: (saleRes.data as SaleDetail | null) ?? null,
      });
    })();
  }, [sessionId]);

  const s = detail?.session;
  const reservation = detail?.reservation;
  const sale = detail?.sale;
  const itemsTotal = detail?.items.reduce((a, i) => a + Number(i.price) * i.quantity, 0) ?? 0;
  const payments: Payment[] = sale
    ? sale.payments?.length
      ? sale.payments
      : [{ method: sale.payment_method, amount: Number(sale.total) }]
    : [];
  const responsible = reservation ?? s;

  return (
    <div className="fixed inset-0 bg-slate-900/60 backdrop-blur-sm flex items-center justify-center z-50 p-4" onClick={onClose}>
      <div className="bg-white rounded-2xl shadow-2xl w-full max-w-2xl max-h-[92vh] flex flex-col" onClick={(e) => e.stopPropagation()}>
        <div className="flex items-center justify-between px-6 py-4 border-b border-slate-100">
          <div className="flex items-center gap-3 min-w-0">
            <span className={`px-2.5 py-1 rounded-lg text-xs font-black tracking-wide ${EVENT_META.jugado.chip}`}>
              {EVENT_META.jugado.label}
            </span>
            <h3 className="text-lg font-bold text-slate-800 truncate">
              {s ? `Turno #${s.session_number} · ${courtLabel(s.court)}` : 'Detalle del turno'}
            </h3>
          </div>
          <button onClick={onClose} className="p-2 rounded-lg hover:bg-slate-100 text-slate-500">
            <X size={20} />
          </button>
        </div>

        {error ? (
          <p className="m-6 bg-red-50 border border-red-200 text-red-700 rounded-xl px-4 py-3 text-sm">{error}</p>
        ) : !s || !detail ? (
          <div className="flex justify-center py-16">
            <Loader2 className="animate-spin text-slate-400" />
          </div>
        ) : (
          <div className="p-6 space-y-6 overflow-y-auto">
            <div className="bg-slate-50 rounded-2xl p-4 flex flex-wrap items-center justify-between gap-4">
              <div>
                <p className="text-[11px] font-bold uppercase text-slate-500">
                  {reservation ? 'Responsable de la reserva' : 'Responsable del turno'}
                </p>
                <p className="text-xl font-bold text-slate-900">{responsible!.titular_name}</p>
                <p className="text-sm text-slate-500">Lote {responsible!.titular_lot || '-'}</p>
              </div>
              <div className="text-right">
                <p className="text-[11px] font-bold uppercase text-slate-500">Total cobrado</p>
                <p className="text-2xl font-bold text-emerald-600">{money(Number(s.total ?? itemsTotal))}</p>
              </div>
            </div>

            <section>
              <p className={sectionTitle}>
                <Clock size={14} /> Horario
              </p>
              <div className="grid grid-cols-2 sm:grid-cols-4 gap-2 text-sm">
                {[
                  ['Fecha', formatDateTime(s.opened_at)],
                  ['Apertura', formatClock(s.opened_at)],
                  ['Cierre', s.closed_at ? formatClock(s.closed_at) : '-'],
                  ['Duración', formatDuration(s.opened_at, s.closed_at)],
                ].map(([label, value]) => (

                    <p className="text-[11px] text-slate-500 uppercase">{label}</p>
                    <p className="font-bold text-slate-800">{value}</p>
                  </div>
                ))}
              </div>
              {reservation && (
                <p className="mt-2 text-sm text-slate-600 flex items-center gap-2">
                  <CalendarCheck size={14} className="text-sky-600" />
                  Reservado de {formatTime(reservation.start_time)} a {formatTime(reservation.end_time)}
                  {reservation.notes && ` · ${reservation.notes}`}
                </p>
              )}
            </section>

            <section>
              <p className={sectionTitle}>
                <Users size={14} /> Jugadores
              </p>
              <ul className="grid sm:grid-cols-2 gap-2">
                {s.players.map((p, i) => (
                  <li key={i} className="flex items-center justify-between gap-2 border border-slate-200 rounded-xl px-3 py-2">
                    <div className="min-w-0">
                      <p className="font-semibold text-slate-800 truncate">
                        J{i + 1}: {p.name}
                      </p>
                      <p className="text-xs text-slate-500">Lote {p.lot || '-'}</p>
                    </div>
                    <span
                      className={`shrink-0 px-2 py-0.5 rounded-full text-[11px] font-bold ${
                        p.is_socio ? 'bg-emerald-100 text-emerald-800' : 'bg-red-100 text-red-700'
                      }`}
                    >
                      {p.is_socio ? 'SOCIO' : 'INVITADO'}
                    </span>
                  </li>
                ))}
              </ul>
            </section>

            <section>
              <p className={sectionTitle}>
                <ShoppingBag size={14} /> Consumos
              </p>
              {detail.items.length === 0 ? (
                <p className="text-sm text-slate-400">Sin consumos.</p>
              ) : (
                <div className="border border-slate-200 rounded-xl divide-y divide-slate-100 text-sm">
                  {detail.items.map((i) => (
                    <div key={i.id} className="flex items-center justify-between gap-3 px-3 py-2">
                      <span className={i.kind === 'invitado' ? 'font-semibold text-red-700' : 'text-slate-800'}>
                        {i.product_name} x{i.quantity}
                        <span className="text-slate-400 font-normal"> · {money(Number(i.price))} c/u</span>
                      </span>
                      <span className="font-semibold">{money(Number(i.price) * i.quantity)}</span>
                    </div>
                  ))}
                  <div className="flex justify-between px-3 py-2 bg-slate-50 font-bold">
                    <span>Total</span>
                    <span>{money(itemsTotal)}</span>
                  </div>
                </div>
              )}
            </section>

            <section>
              <p className={sectionTitle}>
                <Receipt size={14} /> Pago
              </p>
              {sale ? (
                <div className="space-y-2 text-sm">
                  <div className="border border-slate-200 rounded-xl divide-y divide-slate-100">
                    {payments.map((p, i) => (
                      <div key={i} className="flex justify-between px-3 py-2">
                        <span className="font-medium text-slate-700">{PAYMENT_LABELS[p.method] ?? p.method}</span>
                        <span className="font-semibold">{money(Number(p.amount))}</span>
                      </div>
                    ))}
                  </div>
                  {payments.some((p) => p.method === 'cuenta_corriente') && (
                    <p className="bg-amber-50 border border-amber-200 text-amber-800 rounded-xl px-3 py-2">
                      Cargado a cuenta corriente de <strong>{sale.customer_name || '-'}</strong> · Lote {sale.customer_lot || '-'}
                    </p>
                  )}
                  <p className="text-slate-500">
                    Venta {sale.sale_number} · {formatDateTime(sale.created_at)} {formatClock(sale.created_at)} · Cobró: {sale.user_name || '-'}
                  </p>
                </div>
              ) : (
                <p className="text-sm text-slate-500">{s.status === 'cerrada' ? 'Sin cobro ($0).' : 'Turno sin cobrar.'}</p>

import { useEffect, useState } from 'react';

import { CourtReservation } from '../../lib/supabase';
import { EVENT_META, EventType, hhmmOf, minutesOf, nowTimeHHMM, todayISO } from './reservasUtils';

export interface AgendaBlock {
  key: string;
  kind: EventType | 'en_juego' | 'jugado';
  court: number;
  date: string;
  startMin: number;
  endMin: number;
  headline: string;
  detail: string;
  reservation?: CourtReservation;
  sessionId?: string;
}

export interface AgendaColumn {
  key: string;
  label: string;
  highlight?: boolean;
  court: number;
  date: string;
}

interface AgendaTimelineProps {
  columns: AgendaColumn[];
  blocks: AgendaBlock[];
  onBlockClick: (block: AgendaBlock) => void;
  onEmptyClick: (court: number, date: string, time: string) => void;
}

const PX_PER_HOUR = 64;
const END_HOUR = 24;
const DEFAULT_START_HOUR = 7;

export default function AgendaTimeline({ columns, blocks, onBlockClick, onEmptyClick }: AgendaTimelineProps) {
  const [nowMin, setNowMin] = useState(() => minutesOf(nowTimeHHMM()));
  useEffect(() => {
    const t = setInterval(() => setNowMin(minutesOf(nowTimeHHMM())), 60000);
    return () => clearInterval(t);
  }, []);

  const earliest = blocks.reduce((m, b) => Math.min(m, b.startMin), DEFAULT_START_HOUR * 60);
  const startHour = Math.floor(earliest / 60);
  const hours = Array.from({ length: END_HOUR - startHour }, (_, i) => startHour + i);
  const height = (END_HOUR - startHour) * PX_PER_HOUR;
  const top = (min: number) => ((min - startHour * 60) / 60) * PX_PER_HOUR;
  const today = todayISO();

  const handleEmpty = (e: React.MouseEvent<HTMLDivElement>, col: AgendaColumn) => {
    const y = e.clientY - e.currentTarget.getBoundingClientRect().top;
    const minute = startHour * 60 + Math.floor((y / PX_PER_HOUR) * 2) * 30;
    onEmptyClick(col.court, col.date, hhmmOf(minute));
  };

  return (
    <div className="overflow-x-auto">
      <div className="min-w-[640px]">
        <div className="flex sticky top-0 z-20 bg-white border-b border-slate-200">
          <div className="w-14 shrink-0" />
          {columns.map((col) => (
            <div
              key={col.key}
              className={`flex-1 text-center py-3 text-sm font-bold capitalize ${col.highlight ? 'text-sky-700' : 'text-slate-700'}`}
            >
              {col.label}
            </div>
          ))}
        </div>

        <div className="flex relative" style={{ height }}>
          <div className="w-14 shrink-0 relative">
            {hours.map((h) => (
              <span
                key={h}
                className="absolute right-2 -translate-y-1/2 text-[11px] font-semibold text-slate-400"
                style={{ top: top(h * 60) }}
              >
                {h > startHour && `${String(h).padStart(2, '0')}:00`}
              </span>
            ))}
          </div>

          {columns.map((col) => (
            <div
              key={col.key}
              onClick={(e) => handleEmpty(e, col)}
              className="flex-1 relative border-l border-slate-200 cursor-pointer hover:bg-sky-50/40 transition-colors"
            >
              {hours.map((h) => (
                <div key={h} className="absolute inset-x-0 border-t border-slate-100" style={{ top: top(h * 60) }}>
                  <div className="border-t border-dashed border-slate-100" style={{ marginTop: PX_PER_HOUR / 2 }} />
                </div>
              ))}

              {col.date === today && nowMin >= startHour * 60 && (
                <div className="absolute inset-x-0 z-10 pointer-events-none" style={{ top: top(nowMin) }}>
                  <div className="h-0.5 bg-red-500" />
                  <div className="absolute -left-1 -top-1 w-2.5 h-2.5 rounded-full bg-red-500" />
                </div>
              )}

              {blocks
                .filter((b) => b.court === col.court && b.date === col.date)
                .map((b) => {
                  const meta = EVENT_META[b.kind];
                  const blockHeight = Math.max(top(b.endMin) - top(b.startMin), 22);
                  const readOnly = b.kind === 'jugado';
                  const compact = blockHeight <= 34;
                  return (
                    <div
                      key={b.key}
                      role="button"
                      onClick={(e) => {
                        e.stopPropagation();
                        onBlockClick(b);
                      }}
                      title={`${hhmmOf(b.startMin)} - ${hhmmOf(b.endMin)} ${meta.label}\n${b.headline}${b.detail ? `\n${b.detail}` : ''}`}
                      className={`absolute left-1 right-1 rounded-lg border-l-4 px-2 py-1 overflow-hidden shadow-sm text-[11px] leading-tight transition-all cursor-pointer hover:shadow-md hover:z-10 ${meta.block}`}
                      style={{ top: top(b.startMin) + 1, height: blockHeight - 2 }}
                    >
                      <p className="font-bold flex items-center gap-1 truncate">
                        {compact ? (
                          <>
                            {readOnly && <Lock size={10} className="shrink-0 opacity-60" />}
                            <span className="font-black truncate">{b.headline}</span>
                            <span className="font-semibold opacity-70 shrink-0">
                              {hhmmOf(b.startMin)}-{hhmmOf(b.endMin)}
                            </span>
                          </>
                        ) : (
                          <>
                            {hhmmOf(b.startMin)} - {hhmmOf(b.endMin)}
                            <span className="font-black tracking-wide">{meta.label}</span>
                            {readOnly && <Lock size={10} className="shrink-0 opacity-60" />}
                          </>
                        )}
                      </p>
                      {!compact && <p className="truncate font-semibold">{b.headline}</p>}
                      {blockHeight > 50 && b.detail && <p className="truncate opacity-80">{b.detail}</p>}
                    </div>
                  );

            </div>

import { useCallback, useEffect, useMemo, useState } from 'react';

  onCreate: (type: EventType, court: number, date: string, time?: string) => void;
}

const LAST_MINUTE = 23 * 60 + 59;

const CREATE_BUTTONS: { type: EventType; label: string; icon: typeof Trophy }[] = [
  { type: 'reserva', label: 'Registrar reserva', icon: CalendarPlus },
  { type: 'clase', label: 'Registrar clase', icon: GraduationCap },
  { type: 'torneo', label: 'Registrar torneo', icon: Trophy },
  { type: 'mantenimiento', label: 'Mantenimiento', icon: Wrench },
];

const playerLabel = (lot: string, name: string) => `${name}${lot ? ` · Lote ${lot}` : ''}`;

const buildBlocks = (events: CourtReservation[], sessions: CourtSession[]): AgendaBlock[] => {
  const blocks: AgendaBlock[] = [];
  const byId = new Map(events.map((e) => [e.id, e]));
  const nowMin = minutesOf(nowTimeHHMM());
  const today = todayISO();

  for (const e of events) {
    if (e.status !== 'reservada') continue;
    const fallback = e.event_type === 'mantenimiento' ? 'Cancha bloqueada' : '';
    blocks.push({
      key: e.id,
      kind: e.event_type,
      court: e.court,
      date: e.reservation_date,
      startMin: minutesOf(formatTime(e.start_time)),
      endMin: minutesOf(formatTime(e.end_time)),
      headline: eventSummary(e) || fallback,
      detail: e.notes,
      reservation: e,
    });
  }

  for (const s of sessions) {
    const date = localDateOf(s.opened_at);
    const startMin = minutesOf(formatClock(s.opened_at));
    const linked = s.reservation_id ? byId.get(s.reservation_id) : undefined;
    let endMin: number;
    if (s.status === 'abierta') {
      const liveEnd = date === today ? nowMin : LAST_MINUTE;
      endMin = Math.max(liveEnd, linked ? minutesOf(formatTime(linked.end_time)) : 0, startMin + 30);
    } else {
      endMin = s.closed_at && localDateOf(s.closed_at) === date ? minutesOf(formatClock(s.closed_at)) : LAST_MINUTE;
    }
    blocks.push({
      key: s.id,
      kind: s.status === 'abierta' ? 'en_juego' : 'jugado',
      court: s.court,
      date,
      startMin,
      endMin: Math.min(Math.max(endMin, startMin + 1), LAST_MINUTE),
      headline: linked ? playerLabel(linked.titular_lot, linked.titular_name) : playerLabel(s.titular_lot, s.titular_name),
      detail: s.status === 'abierta' ? `Turno #${s.session_number} en curso` : 'Cancha utilizada',
      sessionId: s.id,
    });
  }
  return blocks.sort((a, b) => a.startMin - b.startMin);
};

        <div className="p-4 grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-7 gap-3">
          {days.map((d) => {
            const dayBlocks = blocks.filter((b) => b.date === d);
            return (
              <div
                key={d}
                className={`rounded-xl border p-2 min-h-[120px] ${d === today ? 'border-sky-300 bg-sky-50/40' : 'border-slate-200'}`}
              >
                <button
                  onClick={() => onCreate('reserva', 0, d < today ? today : d)}
                  className="w-full text-left text-xs font-bold uppercase text-slate-500 mb-2 capitalize hover:text-sky-700"
                >
                  {formatDateShort(d)}
                </button>
                <div className="space-y-1.5">
                  {dayBlocks.map((b) => (
                    <button
                      key={b.key}
                      onClick={() => handleBlock(b)}
                      className={`w-full text-left px-2 py-1.5 rounded-lg border-l-4 text-[11px] leading-tight transition-colors ${EVENT_META[b.kind].block}`}
                    >
                      <p className="font-bold flex items-center gap-1">
                        {hhmmOf(b.startMin)}-{hhmmOf(b.endMin)} · C{b.court}
                        {b.kind === 'jugado' && <Lock size={10} className="opacity-60" />}
                      </p>
                      <p className="font-black tracking-wide">{EVENT_META[b.kind].label}</p>
                      <p className="truncate">{b.headline}</p>
                    </button>
                  ))}
                  {dayBlocks.length === 0 && <p className="text-xs text-slate-300">Sin eventos</p>}
                </div>
              </div>
            );
          })}
        </div>
      ) : (
        <div className="p-4">
          <AgendaTimeline
            columns={columns}
            blocks={blocks}
            onBlockClick={handleBlock}
            onEmptyClick={(c, d, time) => onCreate('reserva', c, d, time)}
          />
        </div>
      )}

      {detailId && <TurnoDetailModal sessionId={detailId} onClose={() => setDetailId(null)} />}
    </div>
  );
}

import { Fragment, useEffect, useState } from 'react';
import { ChevronDown, ChevronRight, Loader2 } from 'lucide-react';
import { supabase, CourtReservation, CourtSession, CourtSessionItem } from '../../lib/supabase';
import { PAYMENT_LABELS, PaymentMethod } from '../../lib/processSale';
import { EVENT_META, addDaysISO, eventSummary, formatClock, formatDateTime, formatDuration, formatTime, money, todayISO } from './reservasUtils';

interface SaleRef {
  id: string;
  sale_number: string;
  payments: { method: PaymentMethod; amount: number }[] | null;
  payment_method: PaymentMethod;
}

type Estado = 'Reservada' | 'Abierta' | 'Cerrada' | 'Cancelada';

interface Row {
  key: string;
  id: string;
  court: number;
  date: string;
  apertura: string;
  cierre: string;
  duracion: string;
  estado: Estado;
  session?: CourtSession;
  reservation?: CourtReservation;
}

const estadoStyle: Record<Estado, string> = {
  Reservada: 'bg-sky-100 text-sky-800',
  Abierta: 'bg-amber-100 text-amber-800',
  Cerrada: 'bg-emerald-100 text-emerald-800',
  Cancelada: 'bg-red-100 text-red-700',
};

const ARG_DAY_START = (iso: string) => `${iso}T03:00:00Z`;

export default function TurnosHistory({ refreshKey }: { refreshKey: number }) {
  const [from, setFrom] = useState(addDaysISO(todayISO(), -30));
  const [to, setTo] = useState(todayISO());
  const [rows, setRows] = useState<Row[]>([]);
  const [items, setItems] = useState<CourtSessionItem[]>([]);
  const [sales, setSales] = useState<SaleRef[]>([]);
  const [expanded, setExpanded] = useState<string | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState('');

  useEffect(() => {
    (async () => {
      setLoading(true);
      const [sessionsRes, reservationsRes] = await Promise.all([
        supabase
          .from('court_sessions')
          .select('*')
          .gte('opened_at', ARG_DAY_START(from))
          .lt('opened_at', ARG_DAY_START(addDaysISO(to, 1)))
          .order('opened_at', { ascending: false }),
        supabase
          .from('court_reservations')
          .select('*')
          .gte('reservation_date', from)
          .lte('reservation_date', to)
          .in('status', ['reservada', 'cancelada']),
      ]);
      if (sessionsRes.error || reservationsRes.error) {
        setError('No se pudo cargar el historial.');
        setLoading(false);
        return;
      }
      const sessions = (sessionsRes.data as CourtSession[]) || [];
      const reservations = (reservationsRes.data as CourtReservation[]) || [];
      const sessionIds = sessions.map((s) => s.id);
      const saleIds = sessions.map((s) => s.sale_id).filter((id): id is string => !!id);

      const [itemsRes, salesRes] = await Promise.all([
        sessionIds.length
          ? supabase.from('court_session_items').select('*').in('session_id', sessionIds)
          : Promise.resolve({ data: [] }),
        saleIds.length
          ? supabase.from('sales').select('id, sale_number, payments, payment_method').in('id', saleIds)
          : Promise.resolve({ data: [] }),
      ]);

      const sessionRows: Row[] = sessions.map((s) => ({
        key: s.id,
        id: `#${s.session_number}`,
        court: s.court,
        date: s.opened_at,
        apertura: formatClock(s.opened_at),
        cierre: s.closed_at ? formatClock(s.closed_at) : '-',
        duracion: formatDuration(s.opened_at, s.closed_at),
        estado: s.status === 'abierta' ? 'Abierta' : s.status === 'cerrada' ? 'Cerrada' : 'Cancelada',
        session: s,
      }));
      const reservationRows: Row[] = reservations.map((r) => ({
        key: r.id,
        id: EVENT_META[r.event_type].label,
        court: r.court,
        date: `${r.reservation_date}T${formatTime(r.start_time)}:00-03:00`,
        apertura: formatTime(r.start_time),
        cierre: formatTime(r.end_time),
        duracion: '-',
        estado: r.status === 'cancelada' ? 'Cancelada' : 'Reservada',
        reservation: r,
      }));

      setRows([...sessionRows, ...reservationRows].sort((a, b) => b.date.localeCompare(a.date)));
      setItems((itemsRes.data as CourtSessionItem[]) || []);
      setSales((salesRes.data as SaleRef[]) || []);
      setError('');
      setLoading(false);
    })();
  }, [from, to, refreshKey]);

  const paymentText = (s: CourtSession) => {
    const sale = sales.find((x) => x.id === s.sale_id);
    if (!sale) return s.status === 'cerrada' ? 'Sin cobro ($0)' : '-';
    const list = sale.payments?.length ? sale.payments : [{ method: sale.payment_method, amount: Number(s.total) }];
    return list.map((p) => PAYMENT_LABELS[p.method] ?? p.method).join(' + ');
  };

  return (
    <div className="bg-white rounded-2xl border border-slate-200 shadow-sm">
      <div className="flex flex-wrap items-end gap-3 p-4 border-b border-slate-100">
        <div>
          <label className="block text-xs font-semibold text-slate-500 mb-1">Desde</label>
          <input type="date" value={from} onChange={(e) => e.target.value && setFrom(e.target.value)} className="px-3 py-2 border border-slate-200 rounded-lg text-sm" />
        </div>
        <div>
          <label className="block text-xs font-semibold text-slate-500 mb-1">Hasta</label>
          <input type="date" value={to} onChange={(e) => e.target.value && setTo(e.target.value)} className="px-3 py-2 border border-slate-200 rounded-lg text-sm" />
        </div>
        <p className="text-sm text-slate-500 ml-auto">{rows.length} registros</p>
      </div>

      {error && <p className="p-4 text-sm text-red-600">{error}</p>}
      {loading ? (
        <div className="flex justify-center py-12">
          <Loader2 className="animate-spin text-slate-400" />
        </div>
      ) : (
        <div className="overflow-x-auto">
          <table className="w-full text-sm min-w-[860px]">
            <thead>
              <tr className="text-left text-xs uppercase text-slate-500 bg-slate-50">
                <th className="px-3 py-3 w-8" />
                <th className="px-3 py-3">ID</th>
                <th className="px-3 py-3">Cancha</th>
                <th className="px-3 py-3">Fecha</th>
                <th className="px-3 py-3">Apertura</th>
                <th className="px-3 py-3">Cierre</th>
                <th className="px-3 py-3">Duración</th>
                <th className="px-3 py-3">Titular</th>
                <th className="px-3 py-3 text-right">Total</th>
                <th className="px-3 py-3">Pago</th>
                <th className="px-3 py-3">Estado</th>
              </tr>
            </thead>
            <tbody>
              {rows.length === 0 && (
                <tr>
                  <td colSpan={11} className="text-center py-10 text-slate-400">
                    No hay turnos en el período.
                  </td>
                </tr>
              )}
              {rows.map((row) => {
                const s = row.session;
                const isOpen = expanded === row.key;
                const rowItems = s ? items.filter((i) => i.session_id === s.id) : [];
                const sale = s ? sales.find((x) => x.id === s.sale_id) : undefined;
                return (

import { useCallback, useEffect, useState } from 'react';
import { CalendarDays, CalendarPlus, History, LayoutGrid } from 'lucide-react';
import { supabase, CourtReservation, CourtSession, Shift } from '../../lib/supabase';
import CourtCard from './CourtCard';
import EventModal from './EventModal';
import OpenCourtModal from './OpenCourtModal';
import TurnoView from './TurnoView';
import ReservationsCalendar from './ReservationsCalendar';
import TurnosHistory from './TurnosHistory';
import { COURTS, EventType, formatTime, minutesOf, nowTimeHHMM, todayISO } from './reservasUtils';

type Tab = 'canchas' | 'agenda' | 'historial';

interface EventDraft {
  type: EventType;
  event?: CourtReservation;
  court: number;
  date?: string;
  time?: string;
}

interface OpenDraft {
  court: number;
  reservation?: CourtReservation;
}

export default function Reservas({ shift }: { shift: Shift | null }) {
  const userName = shift?.user_name ?? '';
  const [tab, setTab] = useState<Tab>('canchas');
  const [turnoId, setTurnoId] = useState<string | null>(null);
  const [sessions, setSessions] = useState<CourtSession[]>([]);
  const [totals, setTotals] = useState<Record<string, number>>({});
  const [todayReservations, setTodayReservations] = useState<CourtReservation[]>([]);
  const [eventDraft, setEventDraft] = useState<EventDraft | null>(null);
  const [openDraft, setOpenDraft] = useState<OpenDraft | null>(null);
  const [refreshKey, setRefreshKey] = useState(0);
  const [loadError, setLoadError] = useState('');

  const loadBoard = useCallback(async () => {
    const [sessionsRes, reservationsRes] = await Promise.all([
      supabase.from('court_sessions').select('*').eq('status', 'abierta'),
      supabase
        .from('court_reservations')
        .select('*')
        .eq('reservation_date', todayISO())
        .eq('status', 'reservada')
        .order('start_time'),
    ]);
    if (sessionsRes.error || reservationsRes.error) {
      setLoadError('No se pudo cargar el estado de las canchas.');
      return;
    }
    setLoadError('');
    const open = (sessionsRes.data as CourtSession[]) || [];
    setSessions(open);
    setTodayReservations((reservationsRes.data as CourtReservation[]) || []);

    if (open.length) {
      const { data: items } = await supabase
        .from('court_session_items')
        .select('session_id, price, quantity')
        .in('session_id', open.map((s) => s.id));
      const sums: Record<string, number> = {};
      for (const i of items || []) {
        sums[i.session_id] = (sums[i.session_id] ?? 0) + Number(i.price) * i.quantity;
      }
      setTotals(sums);
    } else {
      setTotals({});
    }
  }, []);

  useEffect(() => {
    loadBoard();
    const timer = setInterval(loadBoard, 30000);
    return () => clearInterval(timer);
  }, [loadBoard, refreshKey]);

  const refresh = () => setRefreshKey((k) => k + 1);

  const courtEvents = (court: number) => {
    const now = minutesOf(nowTimeHHMM());
    const pending = todayReservations.filter((r) => r.court === court && minutesOf(formatTime(r.end_time)) > now);
    const current = pending.find((r) => minutesOf(formatTime(r.start_time)) <= now + 60);
    return { current, next: current ? undefined : pending[0] };
  };

  if (turnoId) {
    return (
      <TurnoView
        sessionId={turnoId}
        shift={shift}
        onBack={() => {
          setTurnoId(null);
          refresh();
        }}
        onClosed={() => {
          setTurnoId(null);
          refresh();
        }}
      />
    );
  }

  const tabs: { id: Tab; label: string; icon: typeof LayoutGrid }[] = [
    { id: 'canchas', label: 'Canchas', icon: LayoutGrid },
    { id: 'agenda', label: 'Agenda', icon: CalendarDays },
    { id: 'historial', label: 'Historial', icon: History },
  ];

  return (
    <div className="space-y-5">
      <div className="flex flex-wrap items-center justify-between gap-3">
        <div className="flex bg-white border border-slate-200 rounded-xl p-1 shadow-sm">
          {tabs.map((t) => (
            <button
              key={t.id}
              onClick={() => setTab(t.id)}
              className={`flex items-center gap-2 px-4 py-2.5 rounded-lg text-sm font-semibold transition-all ${
                tab === t.id ? 'bg-slate-900 text-white shadow' : 'text-slate-600 hover:bg-slate-100'
              }`}
            >
              <t.icon size={16} /> {t.label}
            </button>
          ))}
        </div>
        <button
          onClick={() => setEventDraft({ type: 'reserva', court: 0 })}
          className="flex items-center gap-2 px-4 py-2.5 rounded-xl bg-sky-600 hover:bg-sky-700 text-white font-semibold shadow transition-colors"
        >
          <CalendarPlus size={18} /> Nueva reserva
        </button>
      </div>

      {loadError && <p className="bg-red-50 border border-red-200 text-red-700 rounded-xl px-4 py-3 text-sm">{loadError}</p>}

      {tab === 'canchas' && (
        <div className="grid grid-cols-1 sm:grid-cols-2 xl:grid-cols-4 gap-5">
          {COURTS.map((court) => {
            const session = sessions.find((s) => s.court === court);
            const { current: reservation, next } = session ? { current: undefined, next: undefined } : courtEvents(court);
            return (
              <CourtCard
                key={court}
                court={court}
                session={session}
                sessionTotal={session ? totals[session.id] ?? 0 : 0}
                reservation={reservation}
                nextEvent={next}
                onSchedule={() => setEventDraft({ type: 'reserva', court })}
                onOpen={() => setOpenDraft({ court })}
                onViewReservation={() => reservation && setEventDraft({ type: reservation.event_type, court, event: reservation })}
                onViewTurno={() => session && setTurnoId(session.id)}
              />
            );
          })}
        </div>
      )}

      {tab === 'agenda' && (
        <ReservationsCalendar
          refreshKey={refreshKey}
          onEdit={(event) => setEventDraft({ type: event.event_type, court: event.court, event })}
          onOpenTurno={setTurnoId}
          onCreate={(type, court, date, time) => setEventDraft({ type, court, date, time })}
        />
      )}

      {tab === 'historial' && <TurnosHistory refreshKey={refreshKey} />}

      {eventDraft && (
        <EventModal
          type={eventDraft.type}
          event={eventDraft.event}
          defaultCourt={eventDraft.court}
          defaultDate={eventDraft.date}
          defaultTime={eventDraft.time}
          userName={userName}
          onClose={() => setEventDraft(null)}
          onSaved={() => {
            setEventDraft(null);
            refresh();
          }}
          onOpenCourt={(reservation) => {
            setEventDraft(null);
            setOpenDraft({ court: reservation.court, reservation });
          }}
        />
      )}

      {openDraft && (
        <OpenCourtModal
          court={openDraft.court}
          reservation={openDraft.reservation}
          userName={userName}
          onClose={() => setOpenDraft(null)}
          onDone={(sessionId) => {
            setOpenDraft(null);
            refresh();
            setTurnoId(sessionId);
          }}
        />
      )}
    </div>
  );
}
